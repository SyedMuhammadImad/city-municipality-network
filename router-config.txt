! City Municipality Network — Router Configuration (Cisco 1941)
! Router-on-a-Stick: one subinterface per VLAN for inter-VLAN routing.

enable
configure terminal

interface g0/0.10
encapsulation dot1Q 10
ip address 192.168.10.1 255.255.255.0
no shutdown

interface g0/0.20
encapsulation dot1Q 20
ip address 192.168.20.1 255.255.255.0
no shutdown

interface g0/0.30
encapsulation dot1Q 30
ip address 192.168.30.1 255.255.255.0
no shutdown

interface g0/0.40
encapsulation dot1Q 40
ip address 192.168.40.1 255.255.255.0
no shutdown

interface g0/0.50
encapsulation dot1Q 50
ip address 192.168.50.1 255.255.255.0
no shutdown

interface g0/0
no shutdown
exit
